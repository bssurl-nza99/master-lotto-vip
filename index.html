import re import math from collections import defaultdict, Counter def parse_lottery_raw_text(raw_text): """ อ่านข้อความดิบ ตัดคำ และแยกแยะ ประเภทหวย, วันที่, เลขบน (3 ตัว), เลขล่าง (2 ตัว) โดยแยกประเภทหวยออกจากกันโดยสิ้นเชิง ไม่ปะปนกัน """ lines = [line.strip() for line in raw_text.strip().split('\n') if line.strip()] structured_data = defaultdict(list) i = 0 while i < len(lines): line = lines[i] # ตรวจสอบแพทเทิร์น "ชื่อหวย | วันที่" if '|' in line: parts = line.split('|') lottery_name = parts[0].strip() draw_date = parts[1].strip() # อ่านบรรทัดถัดไปเป็น เลขบน และ เลขล่าง top3 = lines[i+1] if (i+1 < len(lines)) else "" bot2 = lines[i+2] if (i+2 < len(lines)) else "" # กรองเฉพาะบรรทัดที่ออกผลปกติ (ข้ามกรณี "งดออกผล") if top3.isdigit() and bot2.isdigit(): structured_data[lottery_name].append({ 'date': draw_date, 'top3': top3, 'bot2': bot2, 'all_digits': [int(d) for d in (top3 + bot2)] # รวมตัวเลข 0-9 ที่ปรากฏในงวดนั้น }) i += 3 else: i += 1 else: i += 1 return structured_data def calculate_ensemble_top2(draw_history): """ ประมวลผลสถิติของหวยประเภทนั้นๆ (ใช้ 30 งวดล่าสุด) คำนวณผ่าน 2 โมเดลหลัก: 1. Weighted Frequency (ให้น้ำหนักงวดล่าสุดมากกว่า) 2. Gap & Skip Interval (ดูจังหวะรอบการเว้นงวด) """ scores = {d: 0.0 for d in range(10)} total_draws = len(draw_history) if total_draws == 0: return [0, 1] # --- Model 1: Weighted Frequency Analysis (Exponential Decay) --- decay_factor = 0.92 for idx, record in enumerate(draw_history): weight = math.pow(decay_factor, idx) # งวดล่าสุด (idx=0) น้ำหนัก = 1.0 digit_counts = Counter(record['all_digits']) for digit, count in digit_counts.items(): scores[digit] += count * weight * 1.5 # --- Model 2: Gap & Skip Analysis (ระยะเว้นงวด) --- last_seen = {} for idx, record in enumerate(draw_history): for digit in record['all_digits']: if digit not in last_seen: last_seen[digit] = idx # บันทึกระยะห่างจากงวดล่าสุด for digit in range(10): gap = last_seen.get(digit, total_draws) # ให้คะแนนพิเศษกับเลขที่มีระยะเว้นงวดในจังหวะกำลังกลับมา (3-7 งวด) if 3 <= gap <= 7: scores[digit] += 2.5 elif gap == 0: # เลขเพิ่งออกงวดที่แล้ว (มีโอกาสไหลซ้ำ) scores[digit] += 1.0 # จัดอันดับตัวเลข 0-9 จากคะแนนรวม sorted_digits = sorted(scores.items(), key=lambda x: x[1], reverse=True) top_2 = [digit for digit, score in sorted_digits[:2]] return top_2, sorted_digits # ========================================== # ส่วนประมวลผลตัวอย่างข้อมูลที่คุณส่งมา # ========================================== raw_input_data = """[ใส่ข้อความสถิติหวยทั้งหมดที่นี่]""" # 1. อ่านและแยกแยะประเภทหวย parsed_lottery_db = parse_lottery_raw_text(raw_input_data) # 2. ประมวลผลคัดเลือกเลขเด่น Top 2 แยกตามประเภทหวย print("=== ผลการประมวลผลคัดเลือกเลขเด่น (Top 2) แยกตามประเภทหวย ===\n") for lotto_name, history in parsed_lottery_db.items(): top2_digits, full_scores = calculate_ensemble_top2(history) print(f"📌 ประเภทหวย: {lotto_name} (จำนวนข้อมูลที่ประมวลผล: {len(history)} งวด)") print(f"👉 เลขเด่น Top 2 คัดเลือกแล้ว: {top2_digits[0]} และ {top2_digits[1]}") print(f" (คะแนนจัดอันดับ 0-9: {[d for d, s in full_scores]})\n" + "-"*50)
