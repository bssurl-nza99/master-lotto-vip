import numpy as np import pandas as pd class FullProductionLotteryEngine: def __init__(self): """ ระบบประมวลผลสถิติหวยฉบับสมบูรณ์ (Universal Production Engine) รองรับทุกประเภทหวย อ่านข้อมูลสถิติจริง และคำนวณผ่าน Empirical Probability 100% """ self.rolling_window = 30 self.category_thresholds = { 'ลาวพัฒนา': 0.35, 'ฮานอยปกติ': 0.32, 'default': 0.30 } def parse_pdf_statistics_data(self, raw_data_records): """ แปลงข้อมูลดิบจากสถิติ (รูปแบบโครงสร้างจากไฟล์ PDF) ให้อยู่ในรูป DataFrame ที่สะอาดและเรียงลำดับตามเวลาอย่างถูกต้อง """ df = pd.DataFrame(raw_data_records) if df.empty: return pd.DataFrame() df['date'] = pd.to_datetime(df['date']) df = df.sort_values(['category', 'date']).reset_index(drop=True) return df def temporal_isolation(self, df_history, target_date, category): """ กฎการแยกตัดข้อมูลอนาคต (Strict Temporal Isolation) 100% กรองข้อมูลเฉพาะหมวดหมู่ และเฉพาะข้อมูลที่เกิดขึ้นก่อนวันที่เป้าหมายเท่านั้น (T-1 ลงไป) """ target_dt = pd.to_datetime(target_date) subset = df_history[ (df_history['category'] == category) & (df_history['date'] < target_dt) ].copy() return subset def calculate_empirical_weights(self, isolated_subset): """ คำนวณความถี่และน้ำหนักจากข้อมูลจริงย้อนหลัง (Empirical Weights) โดยดึงสถิติตัวเลขที่ออกจริงในอดีตมาแปลงเป็นความน่าจะเป็นทางสถิติ """ if isolated_subset.empty: return {} recent_data = isolated_subset.tail(self.rolling_window) total_rows = len(recent_data) if total_rows == 0: return {} freq_map = {} for _, row in recent_data.iterrows(): actual = str(row.get('actual', '')) if len(actual) >= 2: num = actual[-2:] # ดึงเลข 2 ตัวท้ายมาคำนวณสถิติ freq_map[num] = freq_map.get(num, 0) + 1 prob_map = {k: v / total_rows for k, v in freq_map.items()} return prob_map def evaluate_candidate_board(self, candidate_board, empirical_probs, initial_top2): """ ประเมินและแก้ปัญหา Ranking Boundary (R3-R5 คอขวด) ตรวจสอบว่าเลขในย่าน R3-R5 มี Empirical Probability สูงพอที่จะดึงขึ้นมาติด Top 2 หรือไม่ """ if not candidate_board or len(candidate_board) < 2: return initial_top2, "INVALID_BOARD" scored_candidates = [] for idx, cand in enumerate(candidate_board): prob = empirical_probs.get(str(cand), 0.0) scored_candidates.append({ "number": cand, "original_rank": idx + 1, "empirical_prob": prob }) # จัดลำดับใหม่ตามค่าความน่าจะเป็นเชิงประจักษ์โดยคงความเสถียรของบอร์ด scored_candidates.sort(key=lambda x: x['empirical_prob'], reverse=True) new_top2 = [scored_candidates[0]['number'], scored_candidates[1]['number']] if set(new_top2) != set(initial_top2): return new_top2, "CALIBRATED_FROM_BOUNDARY" return initial_top2, "MAINTAINED_ORIGINAL_TOP2" def run_full_oos_simulation(self, raw_records): """ รันจำลองการทดสอบย้อนหลังแบบเต็มรูปแบบ (Full OOS Simulation) บนชุดข้อมูลสถิติทั้งหมด เพื่อวัดผลความแม่นยำจริงตามเกณฑ์มาตรฐาน """ df = self.parse_pdf_statistics_data(raw_records) if df.empty: return {"status": "ERROR", "message": "No data available"} total_tests = 0 inboard_hits = 0 top2_hits = 0 selection_hits = 0 card_summaries = [] for idx, row in df.iterrows(): target_date = row['date'] category = row['category'] actual_val = str(row['actual']) board = row.get('board', []) if not board or len(board) < 2: continue total_tests += 1 # ตรวจสอบ InBoard (เลขเป้าหมายอยู่ในบอร์ดหรือไม่) inboard_hit = any(item in board for item in [actual_val, actual_val[::-1]]) if inboard_hit: inboard_hits += 1 # ตัดข้อมูลป้องกัน Look-ahead Bias safe_history = self.temporal_isolation(df, target_date, category) # คำนวณ Empirical Probabilities จากข้อมูลจริง empirical_probs = self.calculate_empirical_weights(safe_history) initial_top2 = board[:2] # ตรวจสอบ Top 2 เดิม original_top2_hit = any(item in initial_top2 for item in [actual_val, actual_val[::-1]]) if original_top2_hit: top2_hits += 1 # รันระบบคาลิเบรตดึงเลขจาก Boundary final_top2, action_status = self.evaluate_candidate_board(board, empirical_probs, initial_top2) # ตรวจสอบผลลัพธ์สุดท้าย (Selection Hit) final_hit = any(item in final_top2 for item in [actual_val, actual_val[::-1]]) if final_hit: selection_hits += 1 card_summaries.append({ "date": target_date.strftime('%Y-%m-%d'), "category": category, "actual": actual_val, "final_top2": final_top2, "hit": final_hit, "status": action_status }) # คำนวณเปอร์เซ็นต์ความแม่นยำจริง inboard_pct = round((inboard_hits / total_tests) * 100, 2) if total_tests > 0 else 0.0 top2_pct = round((top2_hits / total_tests) * 100, 2) if total_tests > 0 else 0.0 selection_pct = round((selection_hits / total_tests) * 100, 2) if total_tests > 0 else 0.0 # สร้าง Performance Card สรุปผลการประเมิน performance_card = { "total_tests": total_tests, "inboard_accuracy": f"{inboard_pct}%", "top2_accuracy": f"{top2_pct}%", "selection_accuracy": f"{selection_pct}%", "target_achieved": selection_pct >= 70.0 } return { "performance_card": performance_card, "card_summaries": card_summaries } # ========================================== # ตัวอย่างการใช้งานจริงด้วยชุดข้อมูลสถิติ # ========================================== if __name__ == "__main__": # จำลองชุดข้อมูลสถิติ (รูปแบบโครงสร้างจากไฟล์ PDF) sample_pdf_records = [ {"date": "2026-09-01", "category": "ลาวพัฒนา", "actual": "42", "board": ["10", "20", "42", "55", "99"]}, {"date": "2026-09-03", "category": "ลาวพัฒนา", "actual": "42", "board": ["11", "22", "42", "66", "88"]}, {"date": "2026-09-05", "category": "ลาวพัฒนา", "actual": "88", "board": ["88", "20", "30", "40", "50"]}, {"date": "2026-09-07", "category": "ลาวพัฒนา", "actual": "42", "board": ["05", "19", "42", "77", "90"]}, {"date": "2026-09-10", "category": "ฮานอยปกติ", "actual": "95", "board": ["01", "02", "95", "88", "77"]} ] engine = FullProductionLotteryEngine() result = engine.run_full_oos_simulation(sample_pdf_records) print("=== การ์ดสรุปผลการประเมินระบบ (Production Performance Card) ===") for k, v in result['performance_card'].items(): print(f" - {k}: {v}") print("\n=== ตัวอย่างบันทึกการประมวลผล (Card Summaries) ===") for item in result['card_summaries'][:5]: print(f"วันที่: {item['date']} | หวย: {item['category']} | ออกจริง: {item['actual']} | เลือก Top2: {item['final_top2']} | สถานะถูกรางวัล: {item['hit']} | สถานะบอร์ด: {item['status']}")
