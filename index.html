import random class StaticFormulaEngine: def __init__(self): self.formulas = [] # สร้างคลังสูตรคงที่ตายตัว 100 สูตร (ห้ามปรับแต่งหรือเปลี่ยนโครงสร้างระหว่างรัน) for i in range(1, 101): p1 = random.choice(['H', 'T', 'U', 't', 'u']) p2 = random.choice(['H', 'T', 'U', 't', 'u']) m = random.randint(1, 5) self.formulas.append({ 'id': f'Static-F{i}', 'name': f'สูตรล็อกคงที่ #{i} ({m}x{p1} + {p2})', 'calc': lambda r, p1=p1, p2=p2, m=m: (m * r.get(p1, 0) + r.get(p2, 0)) }) def backtest(self, records): results = [] for f in self.formulas: hits = 0 total = len(records) - 1 if total <= 0: continue # ทำการ Backtest ย้อนหลังงวดต่อวด (ไม่มีการมองเห็นอนาคต) for i in range(total): prev_r = records[i] next_r = records[i+1] try: val = f['calc'](prev_r) except: val = 0 d1 = val % 10 d2 = (d1 + 1) % 10 # ตรวจสอบว่าเลขเด่น d1 หรือ d2 ออกในหลักสิบ/หน่วยของงวดถัดไปหรือไม่ valid_digits = [next_r['T'], next_r['U'], next_r['t'], next_r['u']] if d1 in valid_digits or d2 in valid_digits: hits += 1 win_rate = (hits / total) * 100 if total > 0 else 0 results.append({ 'formula': f['name'], 'hits': hits, 'total': total, 'win_rate': win_rate }) # จัดอันดับสูตรจากความแม่นยำสูงสูด และดึงมาเฉพาะ Top 3 results.sort(key=lambda x: x['win_rate'], reverse=True) return results[:3] if __name__ == "__main__": # จำลองข้อมูลสถิติย้อนหลัง (ตัวอย่าง 30 งวด) sample_records = [] for i in range(30): sample_records.append({ 'H': random.randint(0, 9), 'T': random.randint(0, 9), 'U': random.randint(0, 9), 't': random.randint(0, 9), 'u': random.randint(0, 9) }) engine = StaticFormulaEngine() top3 = engine.backtest(sample_records) print("=== ผลการจัดอันดับ Top 3 สูตรเลขเด่นที่ดีที่สุด (ล็อกสูตรตายตัว) ===") for rank, res in enumerate(top3, 1): print(f"อันดับ {rank}: {res['formula']} | แม่นยำ: {res['win_rate']:.1f}% ({res['hits']}/{res['total']} งวด)")
