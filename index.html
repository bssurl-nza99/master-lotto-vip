from collections import Counter import pandas as pd from sklearn.feature_extraction.text import CountVectorizer from sklearn.naive_bayes import MultinomialNB class LotteryBehaviorAnalyzer: def __init__(self, data): # เตรียมข้อมูลให้อยู่ในรูปแบบ DataFrame # data ควรมีคอลัมน์: ['lottery_type', 'date', 'three_digit', 'two_digit'] self.df = pd.DataFrame(data) self.models = {} self._train_models() def _train_models(self): # แยกฝึกโมเดลเรียนรู้พฤติกรรมแยกตาม "ประเภทหวย" for l_type, group in self.df.groupby('lottery_type'): # ตัวอย่างการใช้ Machine Learning เบื้องต้นเพื่อดูความถี่และแพทเทิร์นเลข # แปลงเลข 3 ตัว และ 2 ตัว เป็น features corpus_three = group['three_digit'].tolist() corpus_two = group['two_digit'].tolist() # เก็บสถิติความถี่พื้นฐานแยกตามประเภทหวย self.models[l_type] = { 'three_freq': Counter(corpus_three), 'two_freq': Counter(corpus_two), 'total_draws': len(group), } def predict_best_numbers(self, lottery_type, top_n=3): """คัดเลือกเลขเจาะ (3 ตัว) และเลขเด่น (2 ตัว) ที่ดีที่สุดสำหรับหวยแต่ละประเภท โดยอ้างอิงจากความถี่และพฤติกรรมในอดีต """ if lottery_type not in self.models: return ( f'ไม่พบข้อมูลสถิติของประเภทหวย: {lottery_type} กรุณาตรวจสอบชื่อประเภทอีกครั้ง' ) model = self.models[lottery_type] # คัดเลือกเลข 3 ตัวยอดฮิต (เลขเจาะ) best_three = model['three_freq'].most_common(top_n) # คัดเลือกเลข 2 ตัวยอดฮิต (เลขเด่น) best_two = model['two_freq'].most_common(top_n) return { 'lottery_type': lottery_type, 'analyzed_draws': model['total_draws'], 'top_three_digits': [ {'number': num, 'count': freq} for num, freq in best_three ], 'top_two_digits': [ {'number': num, 'count': freq} for num, freq in best_two ], } # ========================================== # ตัวอย่างการนำข้อมูลจากไฟล์สถิติมาประมวลผล # ========================================== raw_data = [ # ตัวอย่างข้อมูลหุ้นไทยเช้า[span_1](start_span)[span_1](end_span) {'lottery_type': 'หุ้นไทยเช้า', 'date': '10 ก.ย. 69', 'three_digit': '047', 'two_digit': '42'}, {'lottery_type': 'หุ้นไทยเช้า', 'date': '09 ก.ย. 69', 'three_digit': '027', 'two_digit': '38'}, {'lottery_type': 'หุ้นไทยเช้า', 'date': '8 ก.ย. 69', 'three_digit': '398', 'two_digit': '16'}, # ตัวอย่างข้อมูลลาวประตูชัย[span_2](start_span)[span_2](end_span) {'lottery_type': 'ลาวประตูชัย', 'date': '13 ก.ย. 69', 'three_digit': '036', 'two_digit': '08'}, {'lottery_type': 'ลาวประตูชัย', 'date': '12 ก.ย. 69', 'three_digit': '363', 'two_digit': '34'}, {'lottery_type': 'ลาวประตูชัย', 'date': '11 ก.ย. 69', 'three_digit': '332', 'two_digit': '80'}, ] # เริ่มต้นระบบวิเคราะห์ analyzer = LotteryBehaviorAnalyzer(raw_data) # ทดสอบคัดเลือกเลขสำหรับ "หุ้นไทยเช้า" result_thai = analyzer.predict_best_numbers('หุ้นไทยเช้า', top_n=2) print('ผลวิเคราะห์ หุ้นไทยเช้า:', result_thai) # ทดสอบคัดเลือกเลขสำหรับ "ลาวประตูชัย" result_lao = analyzer.predict_best_numbers('ลาวประตูชัย', top_n=2) print('ผลวิเคราะห์ ลาวประตูชัย:', result_lao)
