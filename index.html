/** * Honest Single-Digit Walk-Forward Predictor & Evaluator * - Zero future leakage (uses strictly draws[0 ... t-1] to predict draw[t]) * - 4-factor Z-score normalized scoring (Short Momentum, Medium Frequency, Bounded Gap, Transition) * - Signal Margin filter to separate high-confidence picks from noisy ties */ function extractFourDigits(draw) { // รองรับทั้งรูปแบบ { top3: "510", bot2: "18" } หรือ { d4: [1, 0, 1, 8] } if (Array.isArray(draw.d4) && draw.d4.length === 4) return draw.d4.map(Number); const t3 = String(draw.top3 || "").padStart(3, "0"); const b2 = String(draw.bot2 || "").padStart(2, "0"); return [Number(t3[1]), Number(t3[2]), Number(b2[0]), Number(b2[1])]; } function zNormalize(arr) { const mean = arr.reduce((a, b) => a + b, 0) / arr.length; const variance = arr.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / arr.length; const std = Math.sqrt(variance); if (std < 1e-9) return arr.map(() => 0); return arr.map((v) => (v - mean) / std); } /** * คำนวณคะแนนเลข 0-9 จากประวัติย้อนหลังเท่านั้น (history เรียงจากเก่า -> ใหม่) */ function scoreDigitsHonest(history, config = {}) { const { wShort = 0.35, // น้ำหนักกระแสระยะสั้น (Decay) wMed = 0.25, // น้ำหนักฐานความถี่ระยะกลาง wGap = 0.25, // น้ำหนักการเว้นรอบแบบจำกัดเพดาน wTrans = 0.15, // น้ำหนักเลขตามจากงวดล่าสุด decay = 0.78, // อัตราลดหลั่นน้ำหนักย้อนหลัง medWindow = 18 // หน้าต่างระยะกลาง } = config; const n = history.length; const parsed = history.map(extractFourDigits); // 1. Short-term Exponential Decay Momentum const shortRaw = Array(10).fill(0); let weightSum = 0; for (let i = n - 1, step = 0; i >= 0 && step < 10; i--, step++) { const w = Math.pow(decay, step); weightSum += w; const uniqueInDraw = new Set(parsed[i]); uniqueInDraw.forEach((d) => { shortRaw[d] += w; }); } if (weightSum > 0) { for (let d = 0; d < 10; d++) shortRaw[d] /= weightSum; } // 2. Medium-term Hit Rate const medSlice = parsed.slice(Math.max(0, n - medWindow)); const medRaw = Array(10).fill(0); medSlice.forEach((digits) => { new Set(digits).forEach((d) => { medRaw[d] += 1 / medSlice.length; }); }); // 3. Bounded Mean-Reversion Gap (พีคที่เว้น 2-4 งวด และลดคะแนนถ้าหลุดยาวเกิน 6 งวด) const gapRaw = Array(10).fill(0); for (let d = 0; d < 10; d++) { let absence = 0; for (let i = n - 1; i >= 0; i--) { if (parsed[i].includes(d)) break; absence++; } if (absence === 0) gapRaw[d] = -0.5; // เพิ่งออกงวดล่าสุด else if (absence === 1) gapRaw[d] = 0.2; else if (absence >= 2 && absence <= 4) gapRaw[d] = 1.0; // โซนคืนค่าเฉลี่ย else if (absence === 5) gapRaw[d] = 0.5; else gapRaw[d] = -0.3; // หลุดยาวเกิน 5 งวด ลดความเสี่ยงเลขดับ } // 4. First-Order Transition (จากเลขเด่น/เลขท้ายของงวดล่าสุด) const transRaw = Array(10).fill(0); if (n >= 4) { const lastDigits = new Set(parsed[n - 1]); let pairCount = 0; for (let i = 0; i < n - 1; i++) { const prevSet = new Set(parsed[i]); const nextSet = new Set(parsed[i + 1]); let overlap = 0; lastDigits.forEach((d) => { if (prevSet.has(d)) overlap++; }); if (overlap > 0) { pairCount += overlap; nextSet.forEach((nd) => { transRaw[nd] += overlap; }); } } if (pairCount > 0) { for (let d = 0; d < 10; d++) transRaw[d] /= pairCount; } } // Normalize ทุกแกนให้เป็น Z-score เพื่อไม่ให้แกนใดกลบแกนอื่น const zShort = zNormalize(shortRaw); const zMed = zNormalize(medRaw); const zGap = zNormalize(gapRaw); const zTrans = zNormalize(transRaw); const ranked = []; for (let d = 0; d < 10; d++) { const totalScore = wShort * zShort[d] + wMed * zMed[d] + wGap * zGap[d] + wTrans * zTrans[d]; ranked.push({ digit: d, score: Number(totalScore.toFixed(4)), breakdown: { short: Number(zShort[d].toFixed(2)), med: Number(zMed[d].toFixed(2)), gap: Number(zGap[d].toFixed(2)), trans: Number(zTrans[d].toFixed(2)) } }); } ranked.sort((a, b) => b.score - a.score); const margin = Number((ranked[0].score - ranked[1].score).toFixed(4)); let confidence = "LOW"; if (margin >= 0.45) confidence = "HIGH"; else if (margin >= 0.20) confidence = "MEDIUM"; return { pick: ranked[0].digit, runnerUp: ranked[1].digit, margin, confidence, ranking: ranked }; } /** * รัน Walk-Forward Backtest แบบซื่อสัตย์ 100% สำหรับหวย 1 ประเภท * @param {Array} draws เรียงลำดับเวลาจากเก่าสุด -> ใหม่สุด * @param {Number} warmup จำนวนงวดขั้นต่ำก่อนเริ่มทาย (แนะนำ 6-8 งวด) */ function runHonestWalkForward(draws, warmup = 6) { if (!Array.isArray(draws) || draws.length <= warmup) { return { error: "ข้อมูลน้อยเกินกว่าจะรัน Walk-Forward อย่างซื่อสัตย์", totalTested: 0 }; } let hitsAll = 0; let testedAll = 0; let hitsHigh = 0; let testedHigh = 0; const logs = []; for (let t = warmup; t < draws.length; t++) { // ใช้เฉพาะข้อมูลก่อนงวด t เท่านั้น const pastOnly = draws.slice(0, t); const actualDigits = extractFourDigits(draws[t]); const pred = scoreDigitsHonest(pastOnly); const isHit = actualDigits.includes(pred.pick); testedAll++; if (isHit) hitsAll++; if (pred.confidence === "HIGH") { testedHigh++; if (isHit) hitsHigh++; } logs.push({ date: draws[t].date || draws[t].date_th || `#${t + 1}`, top3: draws[t].top3, bot2: draws[t].bot2, pick: pred.pick, runnerUp: pred.runnerUp, margin: pred.margin, confidence: pred.confidence, hit: isHit }); } // ทายงวดถัดไปในอนาคต (ใช้ข้อมูลทั้งหมดที่มี) const nextPrediction = scoreDigitsHonest(draws); return { totalDraws: draws.length, warmup, allDraws: { hits: hitsAll, tested: testedAll, winRate: Number(((hitsAll / testedAll) * 100).toFixed(2)), theoreticalBaseline: 34.39 }, highConfidenceOnly: { hits: hitsHigh, tested: testedHigh, winRate: testedHigh > 0 ? Number(((hitsHigh / testedHigh) * 100).toFixed(2)) : null }, nextPrediction, logs }; }
