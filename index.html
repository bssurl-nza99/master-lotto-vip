import pandas as pd import numpy as np class SmartLotteryEngine: def __init__(self, raw_df): """ raw_df ควรมีคอลัมน์: ['lottery_type', 'date', 'three_digit', 'two_digit'] """ self.raw_df = raw_df def clean_and_prepare_data(self): """ กรองข้อมูล: 1. ข้ามแถวที่ 'งดออกผล' หรือข้อมูลไม่สมบูรณ์ 2. เรียงลำดับจาก อดีต -> ปัจจุบัน แบบถูกต้อง 100% 3. แยกประเภทหวยออกจากกันโดยสมบูรณ์ """ df = self.raw_df.copy() # ลบช่องว่างขยะ df['three_digit'] = df['three_digit'].astype(str).str.strip() df['two_digit'] = df['two_digit'].astype(str).str.strip() # กรองวันงดออกผลและค่าว่าง df_clean = df[(df['three_digit'] != 'งดออกผล') & (df['two_digit'] != 'งดออกผล')].copy() df_clean = df_clean.dropna(subset=['three_digit', 'two_digit']) # กรองเฉพาะแถวที่มีตัวเลขครบ 3 หลัก และ 2 หลัก df_clean = df_clean[df_clean['three_digit'].str.isdigit() & df_clean['two_digit'].str.isdigit()] grouped_markets = {} for market_name, group in df_clean.groupby('lottery_type'): # เรียงจาก อดีต -> ปัจจุบัน (Reverse index กรณีสถิติเอาปัจจุบันขึ้นก่อน) sorted_group = group.iloc[::-1].reset_index(drop=True) grouped_markets[market_name] = sorted_group return grouped_markets def check_hit_condition(self, next_row, target_digits): """ เงื่อนไขสำคัญ: - เลขเด่นต้องอยู่ใน หลักสิบบน, หน่วยบน, สิบล่าง, หน่วยล่าง - ถ้าไปอยู่ หลักร้อย (H_top) จะไม่นับว่าเข้าเด็ดขาด (ถือว่าหลุด) """ t_digit = str(next_row['three_digit']).zfill(3) b_digit = str(next_row['two_digit']).zfill(2) # ดึงเฉพาะหลักสิบบน, หน่วยบน, สิบล่าง, หน่วยล่าง valid_positions = [int(t_digit[1]), int(t_digit[2]), int(b_digit[0]), int(b_digit[1])] # ตรวจสอบว่ามีเลขเด่นในตำแหน่งที่อนุญาตหรือไม่ for d in target_digits: if d in valid_positions: return True return False def evaluate_all_markets(self): grouped_data = self.clean_and_prepare_data() all_market_reports = {} for market_name, df_market in grouped_data.items(): if len(df_market) < 2: continue formulas_result = { 'Top_Sum_Formula': [], 'Bot_Sum_Formula': [], 'All_Sum_Formula': [] } for i in range(len(df_market) - 1): curr = df_market.loc[i] nxt = df_market.loc[i+1] t_digit = str(curr['three_digit']).zfill(3) b_digit = str(curr['two_digit']).zfill(2) h_top, s_top, u_top = int(t_digit[0]), int(t_digit[1]), int(t_digit[2]) s_bot, u_bot = int(b_digit[0]), int(b_digit[1]) # 1. สูตรผลรวมบน (+1 แต้ม) top_sum_d1 = (h_top + s_top + u_top) % 10 top_sum_d2 = (top_sum_d1 + 1) % 10 formulas_result['Top_Sum_Formula'].append( self.check_hit_condition(nxt, [top_sum_d1, top_sum_d2]) ) # 2. สูตรผลรวมล่าง (+1 แต้ม) bot_sum_d1 = (s_bot + u_bot) % 10 bot_sum_d2 = (bot_sum_d1 + 1) % 10 formulas_result['Bot_Sum_Formula'].append( self.check_hit_condition(nxt, [bot_sum_d1, bot_sum_d2]) ) # 3. สูตรผลรวม 5 หลัก (+1 แต้ม) all_sum_d1 = (h_top + s_top + u_top + s_bot + u_bot) % 10 all_sum_d2 = (all_sum_d1 + 1) % 10 formulas_result['All_Sum_Formula'].append( self.check_hit_condition(nxt, [all_sum_d1, all_sum_d2]) ) # คำนวณสถิติละเอียด: เข้า/หลุด ติดต่อกันสูงสุด และสถานะปัจจุบัน market_stats = [] for f_name, history in formulas_result.items(): if not history: continue total_draws = len(history) win_count = sum(history) accuracy = round((win_count / total_draws) * 100, 2) # คำนวณ Max Consecutive Win / Loss max_win_streak = 0 max_loss_streak = 0 curr_win = 0 curr_loss = 0 for res in history: if res: curr_win += 1 curr_loss = 0 max_win_streak = max(max_win_streak, curr_win) else: curr_loss += 1 curr_win = 0 max_loss_streak = max(max_loss_streak, curr_loss) # สถานะปัจจุบัน (ย้อนดูรอบล่าสุด) current_streak = 0 last_state = history[-1] for res in reversed(history): if res == last_state: current_streak += 1 else: break current_status_str = f"{'เข้า' if last_state else 'หลุด'} {current_streak} รอบติด" market_stats.append({ 'ชื่อสูตร': f_name, 'อัตราเข้า (%):': accuracy, 'เข้าสูงสุด (ครั้ง)': max_win_streak, 'หลุดสูงสุด (ครั้ง)': max_loss_streak, 'สถานะปัจจุบัน': current_status_str }) all_market_reports[market_name] = pd.DataFrame(market_stats) return all_market_reports #
